# asyncpg is imported by backend/core/db.py, which backend/main.py imports at
# startup (init_db/close_db). It was missing here, so `pip install -r

asyncpg

# ccxt is imported by backend/services/exchange_client.py, which the
# ExecutionAgent imports. It was listed only in requirements-exchange.txt,
# which the Dockerfile never installs, so the container could not start.
# Kept here rather than left split: an execution path that cannot import is
# not an optional feature, it is a broken build.
#
# PINNED to a minimum as of Phase 32. `backend/services/polymarket_client.py`
# needs `ccxt.prediction.polymarket` and the `Prediction*` unified types
# (PredictionOutcome.price, PredictionMarket.marketType/underlying/floorStrike),
# which are recent additions — 4.5.73 is the version they were verified against.
# On an older ccxt the adapter import fails, and the client degrades to
# "unavailable, here is why" rather than breaking startup; pinning is what turns
# that silent degradation into a resolvable install error instead.
ccxt>=4.5.73
